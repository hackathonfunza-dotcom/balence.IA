<script src="https://cdn.jsdelivr.net/npm/@tensorflow/tfjs@4.9.0/dist/tf.min.js"></script>

<script>

(async function(){

  // === CONFIG: pega aquí la URL de tu model.json de Teachable Machine ===

  const MODEL_URL = "AQUI_PONES_TU_MODEL_JSON_URL"; // ejemplo: https://teachablemachine.withgoogle.com/models/XXXXXXXX/model.json

  // Mapa Emoción -> Recomendaciones (personaliza)

  const recommendations = {

    "Feliz": [

      "¡Genial! Mantén la energía: define 1 tarea prioritaria y usa un Pomodoro (25m).",

      "Refuerza: comparte un logro breve con tu equipo (1 min).",

      "Si quieres, guarda este momento y continua con tarea alta prioridad."

    ],

    "Triste": [

      "Pausa 3 minutos: respiración 4-4-4 (inhala 4s, mantiene 4s, exhala 4s).",

      "Haz una microtarea de 5 minutos (algo sencillo) para recuperar control.",

      "Camina 5 minutos y escucha una canción que te guste."

    ],

    "Ansioso": [

      "Respira: 4-4-8 (inhala 4s, sostén 4s, exhala 8s) 3 veces.",

      "Reduce la carga: selecciona 1 tarea urgente y pospón lo demás.",

      "Haz un breve ejercicio de estiramiento de 3 minutos."

    ],

    "Neutral": [

      "Organiza una lista rápida: 3 tareas con prioridad 1-2-3.",

      "Usa un Pomodoro (25/5) para darle ritmo a la jornada.",

      "Haz micro-pausas cada 50 minutos para mantener eficiencia."

    ],

    "Enojado": [

      "Respira y cuenta al 10 lentamente.",

      "Haz 2 minutos de expresión física (estiramientos intensos) fuera del escritorio.",

      "Reformula la tarea: ¿qué pequeño paso puedes completar ahora?"

    ],

    "Default": [

      "No fue posible identificar con confianza. Intenta otra vez en buena iluminación."

    ]

  };

  // Carga modelo

  let model;

  try{

    model = await tmLoadModel(MODEL_URL); // función abajo

  }catch(e){

    console.error("Error cargando modelo:",e);

    document.getElementById('recsArea').innerHTML = "<div class='rec'>No se pudo cargar el modelo. Revisa la URL del modelo.</div>";

    return;

  }

  // DOM

  const startBtn = document.getElementById('startBtn');

  const webcamEl = document.getElementById('webcam');

  const videoArea = document.getElementById('videoArea');

  const labelEl = document.getElementById('label');

  const probEl = document.getElementById('prob');

  const recsArea = document.getElementById('recsArea');

  const snapshotBtn = document.getElementById('snapshotBtn');

  let stream;

  startBtn.onclick = async ()=> {

    try{

      stream = await navigator.mediaDevices.getUserMedia({video:true,audio:false});

      webcamEl.srcObject = stream;

      videoArea.style.display = 'block';

      startBtn.disabled = true;

      // start loop

      detectLoop();

    }catch(e){

      alert("No se pudo acceder a la cámara. Revisa permisos.");

    }

  };

  snapshotBtn.onclick = ()=> runPrediction();

  // detectar continuamente cada 1.2s para no saturar CPU

  async function detectLoop(){

    while(stream){

      await runPrediction();

      await new Promise(r=>setTimeout(r,1200));

    }

  }

  async function runPrediction(){

    try{

      // toma frame del video

      const prediction = await model.predict(webcamEl);

      // prediction es array {className, probability}

      const top = prediction.sort((a,b)=>b.probability-a.probability)[0];

      const label = top.className || "Unknown";

      const prob = Math.round(top.probability*100);

      labelEl.textContent = label;

      probEl.textContent = prob + "%";

      showRecommendations(label, prob);

    }catch(e){

      console.error("pred error",e);

    }

  }

  function showRecommendations(label, prob){

    recsArea.innerHTML = "";

    let key = label;

    if(!recommendations[key]) key = "Default";

    const recs = recommendations[key];

    const title = `<div class="badge"><strong>Detectado:</strong> ${label} — confianza ${prob}%</div>`;

    recsArea.insertAdjacentHTML('beforeend', title);

    recs.forEach(r=> {

      recsArea.insertAdjacentHTML('beforeend', `<div class="rec">${r}</div>`);

    });

    // opcional: botón de "Necesito ayuda"

    recsArea.insertAdjacentHTML('beforeend', `<div style="margin-top:8px"><button onclick="alert('Si necesitas apoyo profesional contacta a recursos humanos o líneas de ayuda locales.')">Necesito ayuda</button></div>`);

  }

  // == Función simple para cargar modelo Teachable Machine TensorFlow.js

  async function tmLoadModel(url){

    // Teachable Machine TF.js format expects model.json and metadata

    // cargamos tf.model y metadata

    const model = await tf.loadGraphModel(url);

    // para compatibilidad con la salida, usamos helper que imita predict sobre video

    return {

      predict: async (videoEl)=>{

        // preprocesa frame a tensor 224x224 (o según tu modelo)

        const tensor = tf.browser.fromPixels(videoEl).resizeNearestNeighbor([224,224]).toFloat().expandDims(0).div(255);

        const out = await model.predict(tensor);

        // out puede ser tensor con probabilidades

        const data = out.arraySync ? out.arraySync()[0] : out[0];

        // intento obtener class names desde metadata (no siempre disponible)

        // Si no hay metadata, usa índices

        const classNames = await tryLoadClassNames(url);

        const arr = Array.from(data).map((p,i)=>({className: classNames[i]||("Clase"+i), probability: p}));

        tf.dispose([tensor,out]);

        return arr;

      }

    };

  }

  async function tryLoadClassNames(modelUrl){

    try{

      // metadata.json típico: ../metadata.json o ./metadata.json

      const base = modelUrl.replace(/model.json$/,'');

      const metaUrl = base + 'metadata.json';

      const res = await fetch(metaUrl);

      if(!res.ok) return null;

      const meta = await res.json();

      // Teachable Machine metadata format:

      if(meta && meta.labels) return meta.labels;

      if(meta && meta.class_names) return meta.class_names;

      if(meta && meta['labels']) return meta['labels'];

      // fallback: build from index

      return null;

    }catch(e){

      return null;

    }

  }

})();
